/// Fatorial com Result.
fn fat(n: i32) -> Result<i32, str> {
    if n < 0 { return Err("n negativo"); }
    if n <= 1 { return Ok(1); }
    return Ok(n * fat(n - 1)?);
}

fn main() {
    match fat(5) {
        Ok(v) => println("5! = " + str::from_int(v)),
        Err(e) => println("erro: " + e),
    }
}
